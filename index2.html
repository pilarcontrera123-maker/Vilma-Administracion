    function procesarExcelCSV(e) {

      const file = e.target.files[0];

      if (!file) return;


      const reader = new FileReader();

      reader.onload = function(evt) {

        const lineas = evt.target.result.split('\n');

        let agregados = 0;


        for (let i = 1; i < lineas.length; i++) {

          const col = lineas[i].split(',');

          if (col.length >= 2 && col[0].trim() !== '') {

            dbProductos.push({

              id: Date.now() + i,

              codigo: col[0].trim().toUpperCase(),

              nombre: col[1].trim(),

              categoria: col[2] ? col[2].trim() : 'Fiesta Largo',

              modalidad: col[3] ? col[3].trim() : 'Alquiler',

              precioVenta: parseFloat(col[4]) || 0,

              precioAlquiler: parseFloat(col[5]) || 0,

              talle: col[6] ? col[6].trim() : 'M',

              busto: parseFloat(col[7]) || null,

              cintura: parseFloat(col[8]) || null,

              cadera: parseFloat(col[9]) || null,

              largo: col[10] ? col[10].trim() : '',

              estadoLimpieza: 'Disponible / Limpio'

            });

            agregados++;

          }

        }

        guardarDB();

        renderTablaStock();

        alert(`¡Listo! Se importaron ${agregados} prendas desde tu archivo Excel/CSV.`);

      };

        id: Date.now(), fechaRegistro: new Date().toLocaleDateString(),

        clienta, dni, telefono, domicilio, email, fechaEvento, observaciones,

        items: [...itemsOrdenActual], total,

        pagos: inicial > 0 ? [{ fecha: new Date().toLocaleDateString(), monto: inicial, medio, facturado: false }] : []

      };


      dbOperaciones.push(nuevaOp);

      guardarDB();


      itemsOrdenActual = [];

      document.getElementById('reg-clienta').value = '';

      document.getElementById('reg-dni').value = '';

      document.getElementById('reg-telefono').value = '';

      document.getElementById('reg-domicilio').value = '';

      document.getElementById('reg-email').value = '';

      document.getElementById('reg-fecha-evento').value = '';

      document.getElementById('reg-observaciones').value = '';

      document.getElementById('valor-ajuste').value = 0;

      document.getElementById('monto-pago-inicial').value = 0;

      

      renderTablaOrden();

      calcularTotales();

      renderHistorial();

      renderCajaMovimientos();

      actualizarAgendaDetalle();

      alert('¡Operación guardada correctamente!');

      switchTab('historial');

    }

    function guardarGasto() {

      const concepto = document.getElementById('gasto-concepto').value.trim();

      const monto = parseFloat(document.getElementById('gasto-monto').value) || 0;

      const tipo = document.getElementById('gasto-tipo').value;

      const medio = document.getElementById('gasto-medio').value;


      if (!concepto || monto <= 0) return alert('Ingresa concepto y monto del gasto.');


      dbGastos.push({ id: Date.now(), fecha: new Date().toLocaleDateString(), concepto, tipo, monto, medio, facturado: true });

      guardarDB();

      document.getElementById('gasto-monto').value = '';

      renderCajaMovimientos();

      alert('Gasto registrado.');

    }


    function renderCajaMovimientos() {

      const filtro = document.getElementById('filtro-caja-tipo').value;

      let movimientos = [];


      dbOperaciones.forEach(op => {

        op.pagos.forEach((p, idx) => {

          movimientos.push({

            id: `ing-${op.id}-${idx}`, fecha: p.fecha, tipo: 'Ingreso',

            concepto: `Cobro: ${op.clienta} (DNI ${op.dni || 'S/D'})`, medio: p.medio,

            monto: p.monto, esIngreso: true, opId: op.id, pagoIdx: idx, facturado: p.facturado || false

          });

        });

      });


      dbGastos.forEach(g => {

        movimientos.push({

          id: `egr-${g.id}`, fecha: g.fecha, tipo: `Gasto ${g.tipo}`,

          concepto: g.concepto, medio: g.medio, monto: g.monto, esIngreso: false, facturado: g.facturado

        });

      });


      const totalIngresos = movimientos.filter(m => m.esIngreso).reduce((s, m) => s + m.monto, 0);

      const totalEgresos = movimientos.filter(m => !m.esIngreso).reduce((s, m) => s + m.monto, 0);

      const ef = movimientos.filter(m => m.esIngreso && m.medio === 'Efectivo').reduce((s, m) => s + m.monto, 0);

      const tr = movimientos.filter(m => m.esIngreso && m.medio === 'Transferencia').reduce((s, m) => s + m.monto, 0);

      const ta = movimientos.filter(m => m.esIngreso && m.medio === 'Tarjeta').reduce((s, m) => s + m.monto, 0);


      document.getElementById('caja-total-ingresos').innerText = `$${totalIngresos.toLocaleString()}`;

      document.getElementById('caja-total-egresos').innerText = `$${totalEgresos.toLocaleString()}`;

      document.getElementById('caja-ganancia-neta').innerText = `$${(totalIngresos - totalEgresos).toLocaleString()}`;

      document.getElementById('caja-efectivo').innerText = `$${ef.toLocaleString()}`;

      document.getElementById('caja-transf').innerText = `$${tr.toLocaleString()}`;

      document.getElementById('caja-tarjeta').innerText = `$${ta.toLocaleString()}`;


      if (filtro === 'ingreso') movimientos = movimientos.filter(m => m.esIngreso);

      if (filtro === 'egreso') movimientos = movimientos.filter(m => !m.esIngreso);


      const tbody = document.getElementById('tabla-caja-body');

      tbody.innerHTML = movimientos.map(m => `

        <tr class="border-b hover:bg-gray-50">

          <td class="p-2.5 text-xs text-gray-500">${m.fecha}</td>

          <td class="p-2.5"><span class="px-2 py-0.5 rounded text-xs font-semibold ${m.esIngreso ? 'bg-green-100 text-green-800' : 'bg-red-100 text-red-800'}">${m.tipo}</span></td>

          <td class="p-2.5 font-medium text-gray-800">${m.concepto}</td>

          <td class="p-2.5 text-xs"><span class="border px-1.5 py-0.5 rounded">${m.medio}</span></td>

          <td class="p-2.5 font-bold ${m.esIngreso ? 'text-green-700' : 'text-red-600'}">${m.esIngreso ? '+' : '-'}$${m.monto.toLocaleString()}</td>

          <td class="p-2.5 text-center">

            ${m.esIngreso ? `

              <button onclick="toggleFacturado(${m.opId}, ${m.pagoIdx})" class="text-xs px-2 py-1 rounded font-bold ${m.facturado ? 'bg-blue-100 text-blue-800 border border-blue-300' : 'bg-gray-100 text-gray-500 border'}">

                ${m.facturado ? '🧾 Facturado' : '⚪ Pendiente'}

              </button>

            ` : '<span class="text-xs text-gray-400">Reg.</span>'}

          </td>

        </tr>

      `).join('');

    }


    function toggleFacturado(opId, pagoIdx) {

      const op = dbOperaciones.find(o => o.id === opId);

      if (op && op.pagos[pagoIdx]) {

        op.pagos[pagoIdx].facturado = !op.pagos[pagoIdx].facturado;

        guardarDB();

        renderCajaMovimientos();

      }

    }


    // --- AGENDA & HISTORIAL ---

    function actualizarAgendaDetalle() {

      const fecha = document.getElementById('filtro-fecha-agenda').value;

      const eventos = dbOperaciones.filter(op => op.fechaEvento === fecha);


      document.getElementById('agenda-contador-dia').innerText = eventos.length;

      document.getElementById('agenda-titulo-fecha').innerText = `Eventos para el ${fecha || 'Día'}`;


      const cont = document.getElementById('agenda-lista-eventos');

      if (eventos.length === 0) {

        cont.innerHTML = `<div class="p-6 text-center text-gray-400 italic border rounded-lg">No hay alquileres agendados para esta fecha.</div>`;

        return;

      }


      cont.innerHTML = eventos.map(op => {

        const abonado = op.pagos.reduce((s, p) => s + p.monto, 0);

        const saldo = op.total - abonado;

        return `

          <div class="border rounded-xl p-4 bg-gray-50 flex flex-col md:flex-row justify-between gap-4">

            <div>

              <span class="font-bold text-base text-purple-900">${op.clienta}</span> <span class="text-xs text-gray-500">(DNI: ${op.dni || 'S/D'} - Tel: ${op.telefono})</span>

              <p class="text-xs text-gray-600 mt-1"><strong>Prendas:</strong> ${op.items.map(i => i.nombre + ' [' + i.codigo + ']').join(', ')}</p>

              ${op.observaciones ? `<p class="text-xs text-amber-800 bg-amber-50 p-2 rounded mt-2"><strong>Ajustes/Obs:</strong> ${op.observaciones}</p>` : ''}

            </div>

            <div class="text-right flex flex-col justify-between">

              <div>

                <span class="text-xs text-gray-500 block">Saldo Pendiente</span>

                <span class="text-base font-bold ${saldo > 0 ? 'text-red-600' : 'text-green-600'}">$${saldo.toLocaleString()}</span>

              </div>

              ${saldo > 0 ? `<button onclick="abrirModalPago(${op.id})" class="mt-2 bg-green-600 text-white text-xs px-3 py-1.5 rounded hover:bg-green-700">Abonar Saldo</button>` : '<span class="text-xs font-bold text-green-700 bg-green-100 px-2 py-1 rounded">COMPLETADO</span>'}

            </div>

          </div>

        `;

      }).join('');

    }

            </td>

          </tr>

        `;

      }).join('');

    }

    function guardarPagoCuota() {
